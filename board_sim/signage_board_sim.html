# SignGuard — SHA-256 + on-chain (Ganache) verify before the screen
def on_update(src, content):
    digest = sha256(content)
    if not registry.verify(content_id, digest):   # SignageRegistry.sol on Ganache
        display.safe_fallback(); buzzer.alarm(); return
    display.render(content)     # only authority-signed content passes
